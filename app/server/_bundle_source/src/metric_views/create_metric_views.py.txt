"""Create the 3 governed metric views over Platinum/Gold."""
import json
import time
from databricks.sdk import WorkspaceClient

CATALOG = "ai_demo_gen"
SCHEMA = "dev_andrea_tardif_demo_influencer_revenue_pipeline_modernized"
WAREHOUSE_ID = "202b674cd44819fd"

w = WorkspaceClient()

DDLS = [
    # mv_creator_roi — sourced from gold_creator_performance_daily (already has the rollup-able fields)
    f"""CREATE OR REPLACE VIEW {CATALOG}.{SCHEMA}.mv_creator_roi
WITH METRICS
LANGUAGE YAML
AS $$
version: 1.1
source: {CATALOG}.{SCHEMA}.gold_creator_performance_daily
comment: "Creator ROI metrics (daily grain, filterable to any time window)"
filter: date >= date_sub(current_date(), 90)
dimensions:
  - name: creator_id
    expr: creator_id
  - name: handle
    expr: handle
  - name: tier
    expr: tier
  - name: platform
    expr: platform
  - name: date
    expr: date
measures:
  - name: attributed_revenue
    expr: SUM(attributed_revenue_usd)
    comment: "Sum of creator-attributed revenue"
  - name: engagements
    expr: SUM(engagements)
  - name: ad_spend
    expr: SUM(ad_spend_usd)
  - name: roi
    expr: SUM(attributed_revenue_usd) / NULLIF(SUM(ad_spend_usd), 0)
    comment: "Creator ROI = attributed revenue / ad spend"
$$
""",
    # mv_campaign_revenue — sourced from gold_campaign_roi
    f"""CREATE OR REPLACE VIEW {CATALOG}.{SCHEMA}.mv_campaign_revenue
WITH METRICS
LANGUAGE YAML
AS $$
version: 1.1
source: {CATALOG}.{SCHEMA}.gold_campaign_roi
comment: "Campaign-level revenue, spend, and ROI"
dimensions:
  - name: campaign_id
    expr: campaign_id
  - name: account_id
    expr: account_id
  - name: brand_name
    expr: brand_name
  - name: status
    expr: status
measures:
  - name: total_revenue
    expr: SUM(total_revenue_usd)
  - name: total_spend
    expr: SUM(total_spend_usd)
  - name: roi
    expr: SUM(total_revenue_usd) / NULLIF(SUM(total_spend_usd), 0)
  - name: active_campaigns
    expr: COUNT(CASE WHEN status='active' THEN 1 END)
$$
""",
    # mv_renewal_health — sourced from platinum_brand_360
    f"""CREATE OR REPLACE VIEW {CATALOG}.{SCHEMA}.mv_renewal_health
WITH METRICS
LANGUAGE YAML
AS $$
version: 1.1
source: {CATALOG}.{SCHEMA}.platinum_brand_360
comment: "Brand renewal health metrics for renewal forecasting"
dimensions:
  - name: account_id
    expr: account_id
  - name: brand_name
    expr: brand_name
  - name: tier
    expr: tier
  - name: renewal_status
    expr: renewal_status
  - name: country
    expr: country
measures:
  - name: renewal_arr
    expr: SUM(renewal_arr_usd)
  - name: ytd_attributed_revenue
    expr: SUM(ytd_attributed_revenue_usd)
  - name: active_campaign_count
    expr: SUM(active_campaigns)
  - name: avg_renewal_health_score
    expr: AVG(renewal_health_score)
  - name: at_risk_arr
    expr: SUM(CASE WHEN renewal_status='up_for_renewal' AND renewal_health_score < 60 THEN renewal_arr_usd ELSE 0 END)
$$
""",
]

for ddl in DDLS:
    name = ddl.split("VIEW ")[1].split("\n")[0]
    print(f"Creating {name}...")
    res = w.statement_execution.execute_statement(
        warehouse_id=WAREHOUSE_ID,
        statement=ddl,
        wait_timeout="30s",
    )
    if res.status.state.value not in ("SUCCEEDED",):
        print(f"  ERROR: {res.status.state.value} — {res.status.error}")
        raise SystemExit(1)
    print(f"  OK")

print("All metric views created.")

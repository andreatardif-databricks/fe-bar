"""
Lakebase Postgres connection helper for the Traackr Influencer Revenue API.

The Databricks Apps platform injects PGHOST / PGUSER / PGDATABASE / PGSSLMODE / PGPORT
when a `postgres` resource binding is declared in app.yaml. The password is a 1-hour
OAuth token we refresh out of band.

Locally / in tests we read PGHOST, PGUSER, PGDATABASE, PGPASSWORD from env.
"""
from __future__ import annotations
import os
import time
import threading
from typing import Optional
import psycopg
from psycopg_pool import ConnectionPool
from databricks.sdk import WorkspaceClient

INSTANCE_NAME = os.environ.get("PG_INSTANCE_NAME", "pulsar-influence-lb")
# Postgres schema where the Lakebase synced tables (synced_brands/campaigns/creators) land.
# Set by app.yaml; defaults to the live demo's dev-mode-prefixed schema name.
SYNCED_SCHEMA = os.environ.get(
    "UC_SCHEMA",
    "dev_andrea_tardif_demo_influencer_revenue_pipeline_modernized",
)
_pool: Optional[ConnectionPool] = None
_pool_lock = threading.Lock()
_token_expiry: float = 0.0
_TOKEN_TTL = 45 * 60  # refresh every 45 min (token is 1h)


def _build_password() -> str:
    """Return the Postgres password — either PGPASSWORD env or a fresh Databricks OAuth token."""
    if os.environ.get("PGPASSWORD"):
        return os.environ["PGPASSWORD"]
    w = WorkspaceClient()
    cred = w.database.generate_database_credential(
        request_id=f"traackr-api-{int(time.time())}",
        instance_names=[INSTANCE_NAME],
    )
    return cred.token


_resolved_host: Optional[str] = None
_resolved_user: Optional[str] = None


def _resolve_host_user() -> tuple[str, str]:
    """If PGHOST/PGUSER aren't injected (legacy Database Instance has no `postgres`
    binding type yet), derive them via the SDK from PG_INSTANCE_NAME."""
    global _resolved_host, _resolved_user
    if _resolved_host and _resolved_user:
        return _resolved_host, _resolved_user
    host = os.environ.get("PGHOST")
    user = os.environ.get("PGUSER")
    if not host or not user:
        w = WorkspaceClient()
        if not host:
            inst = w.database.get_database_instance(name=INSTANCE_NAME)
            host = inst.read_write_dns
        if not user:
            user = w.current_user.me().user_name
    _resolved_host, _resolved_user = host, user
    return host, user


def _conn_kwargs() -> dict:
    host, user = _resolve_host_user()
    return {
        "host": host,
        "port": int(os.environ.get("PGPORT", "5432")),
        "dbname": os.environ.get("PGDATABASE", "dbgen_pulsar_influence"),
        "user": user,
        "password": _build_password(),
        "sslmode": os.environ.get("PGSSLMODE", "require"),
        "application_name": os.environ.get("PGAPPNAME", "pulsar-influence-api"),
    }


def _conn_str() -> str:
    kw = _conn_kwargs()
    return " ".join(f"{k}={v}" for k, v in kw.items())


def _refresh_pool_if_needed():
    global _pool, _token_expiry
    now = time.time()
    if _pool is None or now >= _token_expiry:
        with _pool_lock:
            if _pool is None or now >= _token_expiry:
                if _pool is not None:
                    try:
                        _pool.close()
                    except Exception:
                        pass
                _pool = ConnectionPool(
                    conninfo=_conn_str(),
                    min_size=1,
                    max_size=5,
                    open=True,
                    timeout=10,
                )
                _token_expiry = now + _TOKEN_TTL


def get_pool() -> ConnectionPool:
    _refresh_pool_if_needed()
    assert _pool is not None
    return _pool


def fetch_one(sql: str, params: tuple = ()) -> Optional[dict]:
    with get_pool().connection() as conn:
        with conn.cursor(row_factory=psycopg.rows.dict_row) as cur:
            cur.execute(sql, params)
            return cur.fetchone()


def fetch_all(sql: str, params: tuple = ()) -> list[dict]:
    with get_pool().connection() as conn:
        with conn.cursor(row_factory=psycopg.rows.dict_row) as cur:
            cur.execute(sql, params)
            return cur.fetchall()
